    %% ===== 用户模块 =====
    class User {
        -String id
        -String nickname
        -String avatarUrl
        -String bio
        -String[] tags
        -int creditScore
        -UserStatus status
        +login(credentials) Token
        +updateProfile(profileData) void
        +viewCreditHistory() List~CreditRecord~
        +viewOrders() List~Order~
    }

    class Admin {
        +banUser(userId, reason) void
        +auditContent(targetId, action) void
        +arbitrateDispute(disputeId, result) void
    }

    %% ===== 需求模块 =====
    class Demand {
        -String id
        -String publisherId
        -String title
        -String description
        -RewardType rewardType
        -int rewardPoints
        -String[] tags
        -DemandStatus status
        -Date deadline
        -Location location
        +create(demandData) Demand
        +update(demandData) void
        +delete() void
        +filterAndSort(criteria) List~Demand~
        +getFuzzyLocation() Location
    }

    %% ===== 订单模块 =====
    class Order {
        -String id
        -String demandId
        -String publisherId
        -String helperId
        -OrderStatus status
        -String[] evidenceUrls
        -Date createdAt
        -Date updatedAt
        +accept(helperId) void
        +submitEvidence(evidenceUrls) void
        +confirmCompletion() void
        +initiateDispute(reason) Dispute
        -transitionState(action) void
    }

    class Dispute {
        -String id
        -String orderId
        -String initiatorId
        -String reason
        -DisputeStatus status
        -String judgeResult
        +submitEvidence(evidence) void
    }

    %% ===== 消息模块 =====
    class Conversation {
        -String id
        -String orderId
        -String userAId
        -String userBId
        -String lastMessage
        +create(orderId, userA, userB) Conversation
        +listByUser(userId) List~Conversation~
    }

    class Message {
        -String id
        -String conversationId
        -String senderId
        -String content
        -MessageType contentType
        -Date timestamp
        +send() void
        +markAsRead() void
    }

    %% ===== 信用与评价模块 =====
    class Review {
        -String id
        -String orderId
        -String reviewerId
        -String revieweeId
        -int rating
        -String comment
        -boolean isAnonymous
        +submit() void
        +update() void
    }

    class CreditRecord {
        -String id
        -String userId
        -int change
        -String reason
        -Date createdAt
        +recordChange() void
    }

    %% ===== 积分模块 =====
    class PointsTransaction {
        -String id
        -String userId
        -int amount
        -PointsType type
        -String orderId
        -Date createdAt
        +deduct() void
        +transfer() void
        +refund() void
        +exchange(itemId) void
    }

    %% ===== 管理后台模块 =====
    class Report {
        -String id
        -String reporterId
        -String targetUserId
        -String orderId
        -String reason
        -String[] images
        -ReportStatus status
        +submit() void
        +resolve(result) void
    }

    %% ===== 枚举/值对象 =====
    class Location {
        -double latitude
        -double longitude
        -String description
    }

    %% ========== 关系定义 ==========
    
    %% 继承关系
    Admin --|> User : Inheritance

    Demand "1" --> "0..1" Order : generates >
    Demand "0..1" --> "1" Location : contains >

    Order "1" --> "0..1" Dispute : initiates >
    Order "1" --> "1..2" Review : receives >
    Order "1" --> "0..1" Conversation : creates >

    Conversation "1" *-- "*" Message : contains >

    Admin "1" --> "*" Dispute : arbitrates >
    Admin "1" --> "*" Report : handles >
