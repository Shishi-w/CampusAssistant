classDiagram
    direction TB

    %% ╔═══════════════════════════════════════════════════════════════════════╗
    %% ║  用户模块                                                           ║
    %% ╚═══════════════════════════════════════════════════════════════════════╝

    class Account {
        <<abstract>>
        -String id
        -String studentId
        -String passwordHash
        -String nickname
        -String avatarUrl
        -AccountStatus status
    }

    class User {
        -String bio
        -String[] tags
        -int creditScore
        +updateProfile(profileData) void
        +createDemand() Demand
        +acceptDemand(demandId) Order
        +submitReport(targetUserId, orderId, reason, images) Report
    }

    class Admin {
        -String[] permissions
        +banUser(userId, reason) void
        +auditContent(targetId, action) void
        +arbitrateDispute(disputeId, result) void
        +handleReport(reportId, action) void
    }

    User --|> Account
    Admin --|> Account

    class AuthService {
        +login(studentId, password) Token
        +register(studentId, password) Account
    }
    AuthService ..> Account : authenticates/registers

    class OrderQueryService {
        +findByUserId(userId) List~Order~
    }
    OrderQueryService ..> Order : queries

    class CreditQueryService {
        +findByUserId(userId) List~CreditRecord~
    }
    CreditQueryService ..> CreditRecord : queries


    %% ╔═══════════════════════════════════════════════════════════════════════╗
    %% ║  需求模块                                                           ║
    %% ╚═══════════════════════════════════════════════════════════════════════╝

    class Demand {
        -String id
        -String publisherId
        -String title
        -String description
        -RewardType rewardType
        -int rewardPoints
        -DemandStatus status
        -Date deadline
        -Location location
        -int version
        +update(demandData) void
        +delete() void
        +getFuzzyLocation() Location
    }

    class DemandQueryService {
        +filterAndSort(criteria) List~Demand~
    }
    DemandQueryService ..> Demand : queries

    class Tag {
        -String id
        -String name
    }

    class DemandTag {
        -String id
        -String demandId
        -String tagId
    }

    Demand "1" --> "*" DemandTag : has
    Tag "1" --> "*" DemandTag : applied to


    %% ╔═══════════════════════════════════════════════════════════════════════╗
    %% ║  订单与争议模块                                                     ║
    %% ╚═══════════════════════════════════════════════════════════════════════╝

    class Order {
        -String id
        -String demandId
        -String publisherId
        -String helperId
        -OrderStatus status
        -String[] evidenceUrls
        -Date createdAt
        -Date updatedAt
        -int version
        +accept(helperId) void
        +submitEvidence(evidenceUrls) void
        +confirmCompletion() void
        +initiateDispute(reason) Dispute
        -transitionState(action) void
    }

    class StandardDisputeFactory {
        +create(orderId, initiatorId, reason) Dispute
    }

    Order ..> StandardDisputeFactory : uses

    StandardDisputeFactory ..> Dispute : creates

    class Dispute {
        -String id
        -String orderId
        -String initiatorId
        -String reason
        -DisputeStatus status
        -String judgeResult
        +submitEvidence(evidence) void
    }

    Order "1" --> "0..1" Dispute : initiates

    %% ╔═══════════════════════════════════════════════════════════════════════╗
    %% ║  消息模块                                                           ║
    %% ╚═══════════════════════════════════════════════════════════════════════╝

    class IMessageSender {
        <<interface>>
        +send(message) void
    }

    class Conversation {
        -String id
        -String orderId
        -ConversationStatus status
        +create(orderId) Conversation
    }

    class Message {
        -String id
        -String conversationId
        -String senderId
        -String content
        -MessageType contentType
        -Date timestamp
        -boolean isRead
        +markAsRead() void
    }

    MessageSenderService ..> Message : sends
    Conversation "1" *-- "*" Message : contains
    Conversation "1" *-- "*" ConversationMember : contains
    ConversationMember "*" --> "1" User : includes

    class ConversationMember {
        -String id
        -String conversationId
        -String userId
        -MemberRole role
        -long lastReadMsgId
        +join(conversationId, userId) void
        +markRead(msgId) void
    }

    class MessageSenderService {
        +send(message) void
    }

    MessageSenderService ..|> IMessageSender

    Order "1" --> "0..1" Conversation : creates

    class Review {
        -String id
        -String orderId
        -String reviewerId
        -String revieweeId
        -int rating
        -String comment
        -boolean isAnonymous
        +submit() void
        +update() void
    }

    class CreditRecord {
        -String id
        -String userId
        -int change
        -String reason
        -Date createdAt
    }

    Order "1" --> "1..2" Review : receives


    %% ╔═══════════════════════════════════════════════════════════════════════╗

    class IPointsOperation {
        <<interface>>
        +execute(transaction) void

    class PointsTransaction {
        -String id
        -String userId
        -int amount
        -String orderId
        -Date createdAt

        +execute(transaction) void
    }
    TransferStrategy ..|> IPointsOperation

    class RefundStrategy {
        +execute(transaction) void
    }
    RefundStrategy ..|> IPointsOperation

    class ExchangeStrategy {
        +execute(transaction) void
    }
    ExchangeStrategy ..|> IPointsOperation

        -String targetUserId
        -String orderId
        -String reason
        -String[] images
        -String handlerId
        -ReportStatus status

        +resolve(result) void
    }

    class Location {
        -double latitude
        -double longitude
        -String description
    }
