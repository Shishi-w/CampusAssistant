

## 1. ER 图

## 2.  建表 SQL



### 2.1 认证与用户模块

```sql
-- 认证表：管理登录凭证，与认证方式解耦
CREATE TABLE accounts (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '认证ID',
  student_id VARCHAR(20) NOT NULL COMMENT '学号(全局唯一标识)',
  password_hash VARCHAR(60) DEFAULT NULL COMMENT '密码哈希(BCrypt加密，SSO用户可为空)',
  auth_type TINYINT NOT NULL DEFAULT 1 COMMENT '认证方式: 1-SSO, 2-本地注册',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '账户状态: 1-ACTIVE, 2-BANNED',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_student_id (student_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='认证表';

-- 用户资料表：与认证方式解耦，1:1 关联 account
CREATE TABLE users (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '用户ID',
  account_id BIGINT NOT NULL COMMENT '关联认证ID',
  student_id VARCHAR(20) NOT NULL COMMENT '学号(反范式冗余，减少JOIN)',
  nickname VARCHAR(50) NOT NULL COMMENT '昵称',
  avatar_url VARCHAR(255) DEFAULT NULL COMMENT '头像URL',
  gender TINYINT DEFAULT 0 COMMENT '性别: 0-未知, 1-男, 2-女',
  bio VARCHAR(200) DEFAULT NULL COMMENT '个人简介',
  profile VARCHAR(255) DEFAULT NULL COMMENT '个人简介/专长标签，如"雅思7.5,篮球,配环境"',
  phone VARCHAR(100) DEFAULT NULL COMMENT '手机号(AES-256-GCM加密存储)',
  credit_score INT NOT NULL DEFAULT 60 COMMENT '信用分(反范式冗余，定时任务计算更新)',
  points_balance INT NOT NULL DEFAULT 0 COMMENT '积分余额(反范式冗余，事务保证一致)',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_account_id (account_id),
  UNIQUE KEY uk_student_id (student_id),
  KEY idx_nickname (nickname),
  KEY idx_credit_score (credit_score),
  CONSTRAINT fk_users_account FOREIGN KEY (account_id) REFERENCES accounts(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 管理员表：继承认证，扩展权限
CREATE TABLE admins (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '管理员ID',
  account_id BIGINT NOT NULL COMMENT '关联认证ID',
  nickname VARCHAR(50) NOT NULL COMMENT '昵称',
  avatar_url VARCHAR(255) DEFAULT NULL COMMENT '头像URL',
  role VARCHAR(20) NOT NULL DEFAULT 'MODERATOR' COMMENT '角色: SUPER_ADMIN-超级管理员, MODERATOR-普通管理员',
  permissions JSON DEFAULT NULL COMMENT '权限清单(JSON数组)',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_account_id (account_id),
  CONSTRAINT fk_admins_account FOREIGN KEY (account_id) REFERENCES accounts(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='管理员表';

-- 信用记录表
CREATE TABLE credit_records (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '记录ID',
  user_id BIGINT NOT NULL COMMENT '用户ID',
  `change` INT NOT NULL COMMENT '信用分变动值(正数加，负数减)',
  reason VARCHAR(100) NOT NULL COMMENT '变动原因',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (id),
  KEY idx_user_id (user_id),
  KEY idx_change (`change`),
  KEY idx_reason (reason),
  KEY idx_user_id_created (user_id, created_at DESC),
  CONSTRAINT fk_credit_users FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='信用记录表';
```

### 2.2 需求与订单模块

```sql
-- 标签表
CREATE TABLE tags (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '标签ID',
  name VARCHAR(20) NOT NULL COMMENT '标签名称',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='标签表';

-- 位置表
CREATE TABLE locations (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '位置ID',
  latitude DECIMAL(10,7) DEFAULT NULL COMMENT '纬度(经模糊化处理)',
  longitude DECIMAL(10,7) DEFAULT NULL COMMENT '经度(经模糊化处理)',
  description VARCHAR(50) DEFAULT NULL COMMENT '位置文字描述',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (id),
  KEY idx_latitude (latitude),
  KEY idx_longitude (longitude)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='位置表';

-- 需求表
CREATE TABLE demands (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '需求ID',
  publisher_id BIGINT NOT NULL COMMENT '发布者用户ID',
  location_id BIGINT DEFAULT NULL COMMENT '关联位置ID',
  title VARCHAR(50) NOT NULL COMMENT '标题',
  description VARCHAR(500) NOT NULL COMMENT '详细描述',
  category VARCHAR(20) NOT NULL COMMENT '分类: 快递代取/学习辅导/二手交易等',
  reward_type TINYINT NOT NULL COMMENT '报酬类型: 1-积分, 2-无偿, 3-自定义',
  reward_points INT DEFAULT 0 COMMENT '悬赏积分(reward_type=1时有效)',
  custom_reward VARCHAR(50) DEFAULT NULL COMMENT '自定义报酬说明',
  deadline DATETIME DEFAULT NULL COMMENT '期望完成时间',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 1-待接单, 2-已下架, 3-已接单',
  version INT NOT NULL DEFAULT 0 COMMENT '乐观锁版本号',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (id),
  KEY idx_publisher_id (publisher_id),
  KEY idx_category (category),
  KEY idx_status (status),
  KEY idx_reward_type (reward_type),
  KEY idx_version (version),
  KEY idx_status_created (status, created_at DESC),
  CONSTRAINT fk_demands_publisher FOREIGN KEY (publisher_id) REFERENCES users(id),
  CONSTRAINT fk_demands_location FOREIGN KEY (location_id) REFERENCES locations(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='需求表';

-- 需求标签关联表
CREATE TABLE demand_tags (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '关联ID',
  demand_id BIGINT NOT NULL COMMENT '需求ID',
  tag_id BIGINT NOT NULL COMMENT '标签ID',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_demand_tag (demand_id, tag_id),
  KEY idx_demand_id (demand_id),
  KEY idx_tag_id (tag_id),
  CONSTRAINT fk_dt_demand FOREIGN KEY (demand_id) REFERENCES demands(id) ON DELETE CASCADE,
  CONSTRAINT fk_dt_tag FOREIGN KEY (tag_id) REFERENCES tags(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='需求标签关联表';

-- 订单表
CREATE TABLE orders (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '订单ID',
  demand_id BIGINT NOT NULL COMMENT '关联的需求ID',
  publisher_id BIGINT NOT NULL COMMENT '求助方用户ID(反范式冗余)',
  helper_id BIGINT NOT NULL COMMENT '帮助方用户ID',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 1-进行中, 2-待确认, 3-争议中, 4-已完成, 5-已取消',
  evidence_urls JSON DEFAULT NULL COMMENT '完成凭证图片URL数组',
  version INT NOT NULL DEFAULT 0 COMMENT '乐观锁版本号',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_demand_id (demand_id),
  KEY idx_publisher_id (publisher_id),
  KEY idx_helper_id (helper_id),
  KEY idx_status (status),
  KEY idx_version (version),
  KEY idx_publisher_status (publisher_id, status),
  KEY idx_helper_status (helper_id, status),
  CONSTRAINT fk_orders_demand FOREIGN KEY (demand_id) REFERENCES demands(id),
  CONSTRAINT fk_orders_publisher FOREIGN KEY (publisher_id) REFERENCES users(id),
  CONSTRAINT fk_orders_helper FOREIGN KEY (helper_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单表';

-- 争议表
CREATE TABLE disputes (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '争议ID',
  order_id BIGINT NOT NULL COMMENT '关联订单ID',
  initiator_id BIGINT NOT NULL COMMENT '发起人ID',
  reason VARCHAR(300) NOT NULL COMMENT '争议原因',
  publisher_evidence JSON DEFAULT NULL COMMENT '求助方证据截图',
  helper_evidence JSON DEFAULT NULL COMMENT '帮助方证据截图',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 1-待仲裁, 2-已解决',
  judge_result VARCHAR(200) DEFAULT NULL COMMENT '仲裁结果',
  judge_id BIGINT DEFAULT NULL COMMENT '仲裁管理员ID',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (id),
  KEY idx_order_id (order_id),
  KEY idx_initiator_id (initiator_id),
  KEY idx_status (status),
  KEY idx_judge_id (judge_id),
  CONSTRAINT fk_disputes_order FOREIGN KEY (order_id) REFERENCES orders(id),
  CONSTRAINT fk_disputes_initiator FOREIGN KEY (initiator_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='争议表';
```

### 2.3 消息与会话模块


```sql
-- 会话表
CREATE TABLE conversations (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '会话ID',
  order_id BIGINT NOT NULL COMMENT '关联订单ID',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 1-ACTIVE, 2-ARCHIVED',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_order_id (order_id),
  KEY idx_status (status),
  CONSTRAINT fk_conv_order FOREIGN KEY (order_id) REFERENCES orders(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='会话表';

-- 会话成员表
CREATE TABLE conversation_members (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '成员ID',
  conversation_id BIGINT NOT NULL COMMENT '会话ID',
  user_id BIGINT NOT NULL COMMENT '用户ID',
  role TINYINT NOT NULL DEFAULT 1 COMMENT '角色: 1-OWNER, 2-PARTICIPANT',
  last_read_msg_id BIGINT DEFAULT 0 COMMENT '已读消息位点',
  joined_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '加入时间',
  PRIMARY KEY (id),
  UNIQUE KEY uk_conv_user (conversation_id, user_id),
  KEY idx_conversation_id (conversation_id),
  KEY idx_user_id (user_id),
  CONSTRAINT fk_cm_conversation FOREIGN KEY (conversation_id) REFERENCES conversations(id) ON DELETE CASCADE,
  CONSTRAINT fk_cm_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='会话成员表';

-- 消息表
CREATE TABLE messages (
  id BIGINT NOT NULL AUTO_INCREMENT COMMENT '消息ID',
  conversation_id BIGINT NOT NULL COMMENT '会话ID',
  sender_id BIGINT NOT NULL COMMENT '发送者ID',
  content VARCHAR(500) NOT NULL COMMENT '消息内容/图片URL',
  content_type TINYINT NOT NULL DEFAULT 1 COMMENT '内容类型: 1-文本, 2-图片',
  is_read TINYINT NOT NULL DEFAULT 0 COMMENT '是否已读: 0-未读, 1-已读',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '发送时间',
  PRIMARY KEY (id),
  KEY idx_conversation_id (conversation_id),
  KEY idx_sender_id (sender_id),
  KEY idx_created_at (created_at),
  KEY idx_conversation_time (conversation_id, created_at DESC),
  CONSTRAINT fk_msg_conversation FOREIGN KEY (conversation_id) REFERENCES conversations(id) ON DELETE CASCADE,
  CONSTRAINT fk_msg_sender FOREIGN KEY (sender_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='聊天消息表';
```

## 3. 索引设计说明



| 表名                        | 索引字段                          | 索引类型 | 为什么需要这个索引？                                            |
| ------------------------- | ----------------------------- | ---- | ----------------------------------------------------- |
| **accounts**              | `student_id`                  | 唯一索引 | 登录认证高频查询，必须唯一且极速响应。                                   |
| **users**                 | `account_id`                  | 唯一索引 | 1:1 关联认证表，防止一个 account 绑定多个用户资料。                      |
| **users**                 | `student_id`                  | 唯一索引 | 学号是登录和唯一标识的核心，必须保证全局唯一且支持等值精确查询。                      |
| **users**                 | `nickname`                    | 普通索引 | 支持按昵称搜索用户，提升用户检索体验。                                   |
| **users**                 | `credit_score`                | 普通索引 | 支持按信用分排序或筛选（如高信用用户优先推荐）。                              |
| **admins**                | `account_id`                  | 唯一索引 | 1:1 关联认证表，防止一个 account 绑定多个管理员身份。                     |
| **tags**                  | `name`                        | 唯一索引 | 标签名称必须唯一，且支持按名称精确匹配查询，标签检索核心。                         |
| **locations**             | `latitude`                    | 普通索引 | 支持基于纬度的范围查询（配合经度进行LBS附近需求搜索）。                         |
| **locations**             | `longitude`                   | 普通索引 | 支持基于经度的范围查询。                                          |

> ### 一、是否满足第三范式 (3NF) ？
>
>
>
> 结论：存在刻意的反范式冗余（合理），已无非刻意的传递依赖需修正。
>
>
>
> #### 1. 刻意的反范式冗余（保留，但需注意一致性）
>
>
>
> * `users.credit_score` 和 `users.points_balance`：可以通过 `credit_records` 和 `points_transactions` 聚合计算得出，违反了 3NF。但为了防止高频查询时进行全表 JOIN 和 SUM 运算，这种冗余是必要且合理的。需通过 Spring Boot 的 `@Transactional` 严格保证流水表插入与余额更新的原子性。
>
> * `users.student_id`：已通过 `account_id` 关联 `accounts` 表获取学号，但用户主页高频展示学号，反范式冗余可避免额外 JOIN。
>
> * `orders.publisher_id`：可以通过 `demand_id` 关联 `demands` 表推导出发布者，存在传递依赖。保留是为了查询“我参与的订单”时避免再次 JOIN `demands` 表。
>
> * `reviews.reviewee_id`：`order_id` + `reviewer_id` 已经可以推导出被评价者（订单的另一方），存在传递依赖。保留是为了快速查询“我收到的评价”，避免复杂的 JOIN 推导。
>
> #### 2. JSON 字段的使用合理性
>
>
>
> * `orders.evidence_urls`、`disputes.publisher_evidence` / `helper_evidence`、`reports.images`、`admins.permissions` 使用了 JSON 类型。
>
> * 在 MySQL 8.0 中，JSON 是合法的数据类型。以上字段均用于“存取整个数组/对象”，不做元素级检索或反向搜索，使用 JSON 完全合理。
>
> * `demands.tags` 已从 JSON 拆分为 `TAGS` + `DEMAND_TAGS` 多对多关系表，解决了标签搜索的性能问题。
>

> ### 二、索引设计是否合理？
>
>
>
> 结论：索引体系完整，覆盖所有高频查询路径。
>
>
>
> 1. **会话查询优化（已解决）**
>
>
>
>    * 通过 `conversation_members` 表，彻底消除了 `WHERE user_a_id = ? OR user_b_id = ?` 的 OR 查询瓶颈。
>
>    * `idx_user_id` 单列索引完美覆盖“我的会话列表”查询，且天然支持未来扩展群聊。
>
> 2. **标签搜索优化（已解决）**
>
>
>
>    * `demands.tags` 从 JSON 拆分为 `TAGS` + `DEMAND_TAGS` 多对多关系表。
>
>    * `idx_tag_id` 支撑按标签反查需求的核心功能，标准 B+Tree 索引高效可靠。
>
> 3) **举报表索引（已补充）**
>
>
>
>    * 已添加 `idx_target_user_status (target_user_id, status)` 联合索引。
>
>    * 覆盖管理员高频查询“某用户待处理的举报”，实现覆盖索引避免回表。
>
> 4) **冗余索引说明**
>
>
>
>    * `orders` 表中 `idx_publisher_id` 与 `idx_publisher_status`、`idx_helper_id` 与 `idx_helper_status` 存在前缀重叠。MySQL 8.0 优化器在索引下推条件下可复用联合索引前缀，但从严格优化角度，单列索引可被联合索引替代。当前保留两者以兼容更广泛的查询模式。
>
>

> ### 三、是否有潜在的性能瓶颈？
>
>
>
> 结论：原有两处严重性能瓶颈已通过架构重构解决，当前仅需关注消息表长期增长。
>
>
>
> #### 瓶颈 1：高并发接单的锁竞争（超卖问题）
>
>
>
> * **场景**：SRS 要求 FR-O01 “同一需求在同一时间只能被一个帮助方承接”。
>
> * **解决方案**：`demands.version` 乐观锁 + `orders.demand_id` 唯一索引双重保障。
>
> * **接单逻辑**：先更新需求表 `UPDATE demands SET status = 3, version = version + 1 WHERE id = ? AND version = ? AND status = 1;`，判断返回的受影响行数，如果为 1 才插入 `orders` 记录。这利用了数据库行级写锁的排他性，将并发冲突拦截在更新阶段，极大减轻订单表的插入压力。
>
>
>
> #### 瓶颈 2：会话列表的查询性能（OR 逻辑）
>
>
>
> * **场景**：用户打开消息页，查询自己的会话列表。
>
> * **解决方案**：引入 `conversation_members` 关系表（聊天系统的标准设计）。
>
> * **收益**：查询我的会话只需 `WHERE user_id = ?`，完美命中索引；天然支持未来扩展群聊；能把“未读消息数”的计算分摊到成员表上，避免每次扫描 `messages` 表。
>
>
>
> #### 瓶颈 3：消息表长期增长
>
>
>
> * **场景**：`messages` 表随聊天量线性增长。
>
> * **风险**：无时间范围分区，单表可能膨胀到千万级。
>
> * **缓解**：`idx_conversation_time (conversation_id, created_at)` 联合索引可支撑千万级数据的聊天记录拉取。课程项目场景下当前设计完全可行；若真实上线，需考虑按时间分区或归档策略。
>
>

| **标签查询**  | 无                                                  | `demand_tags(tag_id)` 及唯一联合索引                                                                    | 支撑标签搜索这一核心功能，确保通过标签反查需求列表时命中索引。                                            |
| **举报查询**  | 无                                                  | `reports(target_user_id, status)` 联合索引                                                           | 管理员高频查询“某用户待处理的举报”，联合索引可实现覆盖索引，避免回表。                                       |
| **消息记录**  | `conversation_id, created_at` 联合索引                 | `conversation_id` 单列索引 + `created_at` 单列索引 + `conversation_id, created_at` 联合索引                  | 三层索引覆盖：按会话查、按时间查、按会话+时间排序拉取历史记录。                                           |
| **乐观锁索引** | 无                                                  | `demands.version`、`orders.version`                                                               | 乐观锁更新时作为 WHERE 条件，建立索引可提升高并发更新时的定位速度。                                      |
| **唯一约束**  | `orders.demand_id`、`reviews.order_id, reviewer_id` | 增加 `conversation_members(conversation_id, user_id)`、`demand_tags(demand_id, tag_id)`、`tags.name` | 通过数据库唯一索引在底层拦截脏数据（如重复加群、重复打标签），比代码层拦截更可靠。                                  |

>
> ### 四、设计理念的转变
>
>
>
> 1. **从“能跑就行”到“抗高并发”**：引入乐观锁替代隐式的锁表，拆分会话表消除 OR 索引失效，都是为高并发接单和聊天做准备。
>
> 2. **从“严格遵守范式”到“适度反范式”**：明确标记了 `points_balance`、`student_id`、`publisher_id`、`reviewee_id` 四个冗余字段，用极小的数据一致性风险，换取了核心列表查询避免多表 JOIN 的巨大性能收益。
>
> 3) **从“单体大表”到“职责分离”**：认证与资料拆分、地理位置独立、标签多对多关联，使得各表职责单一，便于后续独立优化和扩展。
